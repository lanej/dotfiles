#### Long commands

**Key:** `P`

```sh
if-shell -F '#{==:#{session_name},popup}' {
    detach-client
} {
    popup -E -w 90% -h 90% 'tmux attach -t popup || tmux new -s popup'
}
```

**Key:** `c-f`

```sh
display-popup -E 'win=$(tmux list-windows -F "#{window_bell_flag}|#{window_activity_flag}|#{window_index}|#{window_name} #{?window_bell_flag,[needs input] ,#{?window_activity_flag,[activity] ,}}" | sort -t"|" -k1,1 -rn -k2,2 -rn -k3,3 -n | fzf --height=100% --reverse --header="Windows (current session)" --color=pointer:#d787d7,header:#d787d7,hl+:#ffafaf --delimiter="|" --with-nth=3.. --preview "echo {} | cut -d\"|\" -f3 | xargs -I {} tmux capture-pane -ep -t {} -S -30 2>/dev/null || echo \"Preview unavailable\"" --preview-window=right:60% | cut -d"|" -f3 || true); [ -n "$win" ] && tmux select-window -t "$win"; exit 0'
```

**Key:** `c-w`

```sh
display-popup -E 'sess=$(tmux list-sessions -F "#{session_name}: #{session_windows} windows#{?session_attached, (attached),}" | fzf --height=100% --reverse --header="Sessions" --color=pointer:#d787d7,header:#d787d7,hl+:#ffafaf --preview "echo \"Windows in session {1}:\"; tmux list-windows -t {1} -F \"  #I: #W\" 2>/dev/null; echo \"\"; echo \"Current pane content:\"; tmux capture-pane -ep -t {1} -S -20 2>/dev/null || echo \"Preview unavailable\"" --preview-window=right:60% | sed "s/:.*//g" || true); [ -n "$sess" ] && tmux switch-client -t "$sess"; exit 0'
```

**Key:** `H`

```sh
run-shell 'tmux new-session -d -s later 2>/dev/null; sess="#{session_name}"; tmux move-window -s "$sess" -t later:; tmux move-window -r -t "$sess"'
```
